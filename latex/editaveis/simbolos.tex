\begin{simbolos}
  \item[MS] \textit{Mutation score}, razão entre mutantes mortos e mutantes não equivalentes, em porcentagem: $MS = \frac{K}{M - E} \times 100$
  \item[K] Número de mutantes mortos
  \item[M] Número total de mutantes gerados
  \item[E] Número de mutantes equivalentes
  \item[MSp] \textit{Mutation score} obtido com os mutantes selecionados
  \item[MSa] \textit{Mutation score} obtido com todos os mutantes
  \item[RE] Erro relativo do \textit{mutation score}: $RE = \frac{|MSp - MSa|}{MSa}$
  \item[n] Número de estudos considerados
  \item[$|\,|$] Valor absoluto
  \item[$\times$] Multiplicação
  \item[\%] Porcentagem
  \item[MB] \textit{Megabyte}
\end{simbolos}
