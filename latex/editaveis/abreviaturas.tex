\begin{siglas}
\item[AAM] Operador de mutação do MAEWU que adiciona um atributo a um elemento web
  \item[ABS] \textit{Absolute Value Insertion} (inserção de valor absoluto), operador de mutação
  \item[ACM] \textit{Association for Computing Machinery}
  \item[ADM] Operador de mutação do MAEWU que apaga um atributo de um elemento web
  \item[AMM] Operador de mutação do MAEWU que modifica um atributo de um elemento web
  \item[AOR] \textit{Arithmetic Operator Replacement} (substituição de operador aritmético), operador de mutação
  \item[API] \textit{Application Programming Interface} (interface de programação de aplicações)
  \item[CDM] Operador de mutação do MAEWU que apaga o conteúdo de um elemento web
  \item[CIM] Operador de mutação do MAEWU que insere conteúdo em um elemento web
  \item[CMM] Operador de mutação do MAEWU que modifica o conteúdo de um elemento web
  \item[COR] \textit{Conditional Operator Replacement} (substituição de operador condicional), operador de mutação
  \item[CPU] \textit{Central Processing Unit} (unidade central de processamento)
  \item[CSS] \textit{Cascading Style Sheets}
  \item[DOM] \textit{Document Object Model}
  \item[EHAM] Operador de mutação do MAEWU que adiciona um manipulador de evento
  \item[EHDM] Operador de mutação do MAEWU que apaga um manipulador de evento
  \item[EHMM] Operador de mutação do MAEWU que modifica um manipulador de evento
  \item[EJB] \textit{Enterprise JavaBeans}
  \item[GQM] \textit{Goal Question Metric}
  \item[HTML] \textit{HyperText Markup Language}
  \item[JDK] \textit{Java Development Kit}
  \item[JSP] \textit{JavaServer Pages}
  \item[LCR] \textit{Logical Connector Replacement} (substituição de conector lógico), operador de mutação
  \item[LLM] \textit{Large Language Model} (modelo de linguagem de grande porte)
  \item[MAEWU] \textit{Mutation analysis framework for E2E web testing}, framework de análise de mutação de Yandrapally e Mesbah (2021)
  \item[PICO] \textit{Population, Intervention, Comparison, Outcome} (população, intervenção, comparação e resultados)
  \item[RIP] \textit{Reachability, Infection, Propagation} (alcance, infecção e propagação)
  \item[ROR] \textit{Relational Operator Replacement} (substituição de operador relacional), operador de mutação
  \item[RSL] Revisão Sistemática da Literatura
  \item[SBR] \textit{Statement Block Removal} (remoção de bloco de instruções), operador de mutação
  \item[SCM] Operador de mutação do MAEWU que altera a cor de um elemento web
  \item[SDL] \textit{Statement Deletion} (remoção de instrução), operador de mutação
  \item[SIGSOFT] \textit{Special Interest Group on Software Engineering} da ACM
  \item[SIR] \textit{Software-artifact Infrastructure Repository}
  \item[SPM] Operador de mutação do MAEWU que altera a posição de um elemento web
  \item[SQ] \textit{Secondary Question} (questão de pesquisa secundária)
  \item[SSM] Operador de mutação do MAEWU que altera o tamanho de um elemento web
  \item[SVM] Operador de mutação do MAEWU que altera a visibilidade de um elemento web
  \item[SVR] \textit{Scalar Variable Replacement} (substituição de variável escalar), operador de mutação
  \item[TDM] Operador de mutação do MAEWU que apaga um elemento da árvore DOM
  \item[TIM] Operador de mutação do MAEWU que insere um elemento na árvore DOM
  \item[TMM] Operador de mutação do MAEWU que move um elemento na árvore DOM
  \item[UOI] \textit{Unary Operator Insertion} (inserção de operador unário), operador de mutação
  \item[URL] \textit{Uniform Resource Locator}
  \item[WS-BPEL] \textit{Web Services Business Process Execution Language}
\end{siglas}
